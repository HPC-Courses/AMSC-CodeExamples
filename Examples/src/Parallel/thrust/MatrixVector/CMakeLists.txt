cmake_minimum_required(VERSION 3.24)
project(APSCMatrixVector VERSION 1.0 LANGUAGES CXX)

set(APSC_THRUST_BACKEND CUDA CACHE STRING "Thrust backend: CUDA, CPP, OMP, TBB")
set_property(CACHE APSC_THRUST_BACKEND PROPERTY STRINGS CUDA CPP OMP TBB)
if(NOT APSC_THRUST_BACKEND MATCHES "^(CUDA|CPP|OMP|TBB)$")
  message(FATAL_ERROR "APSC_THRUST_BACKEND must be CUDA, CPP, OMP, or TBB")
endif()
if(APSC_THRUST_BACKEND STREQUAL "CUDA")
  # GPU selection: pass -DCMAKE_CUDA_ARCHITECTURES=89 for your Ada GPU.
  # native detects the local GPU; an explicit value also works without a GPU.
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
    set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "GPU compute capability, e.g. 89, or native")
  endif()
  enable_language(CUDA)
endif()
set(CCCL_ROOT "" CACHE PATH "CCCL source tree or installation prefix")
find_package(Thrust CONFIG REQUIRED HINTS
  "${CCCL_ROOT}/lib/cmake/thrust" "${CCCL_ROOT}/thrust/thrust/cmake"
  "/usr/local/cuda/lib64/cmake/thrust")
thrust_create_target(apsc_thrust HOST CPP DEVICE ${APSC_THRUST_BACKEND})
add_library(apsc_matrix_vector INTERFACE)
add_library(apsc::MatrixVector ALIAS apsc_matrix_vector)
target_include_directories(apsc_matrix_vector INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(apsc_matrix_vector INTERFACE cxx_std_20)
if(APSC_THRUST_BACKEND STREQUAL "CUDA")
  target_compile_features(apsc_matrix_vector INTERFACE cuda_std_20)
endif()
target_link_libraries(apsc_matrix_vector INTERFACE apsc_thrust)

option(APSC_BUILD_EXAMPLES "Build the sparse matrix example" ${PROJECT_IS_TOP_LEVEL})
option(APSC_BUILD_TESTS "Build numerical and API tests (requires Eigen3)" ${PROJECT_IS_TOP_LEVEL})
option(APSC_BUILD_BENCHMARKS "Build the CPU/CUDA sparse benchmark" OFF)
function(apsc_add_program target source)
  add_executable(${target} "${source}")
  target_link_libraries(${target} PRIVATE apsc::MatrixVector)
  if(APSC_THRUST_BACKEND STREQUAL "CUDA")
    set_source_files_properties("${source}" PROPERTIES LANGUAGE CUDA)
    set_target_properties(${target} PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
  endif()
endfunction()
if(APSC_BUILD_BENCHMARKS)
  apsc_add_program(sparse_benchmark benchmarks/sparse_benchmark.cpp)
  target_compile_definitions(sparse_benchmark PRIVATE
    APSC_BENCHMARK_BACKEND="${APSC_THRUST_BACKEND}"
    APSC_BENCHMARK_BUILD_TYPE="$<CONFIG>")
endif()
if(APSC_BUILD_EXAMPLES)
  apsc_add_program(sparse_example examples/sparse_example.cpp)
endif()
if(APSC_BUILD_TESTS)
  enable_testing()
  find_package(Eigen3 3.3 REQUIRED NO_MODULE)
  apsc_add_program(sparse_tests tests/sparse_tests.cpp)
  target_link_libraries(sparse_tests PRIVATE Eigen3::Eigen)
  # Eigen supplies only the host reference calculation, never GPU kernels.
  target_compile_definitions(sparse_tests PRIVATE EIGEN_NO_CUDA)
  add_test(NAME sparse_tests COMMAND sparse_tests)
  set_tests_properties(sparse_tests PROPERTIES TIMEOUT 120)
endif()
